# Region Stats shared numeric values

# Wealth is exposed to script as controlled tax base. Dividing by local control
# recovers the same raw location wealth displayed by the Location Wealth map
# mode. The minimum prevents division by zero in locations with no control.
eap_location_wealth = {
	value = location_tax_base
	divide = {
		value = local_control
		min = 0.01
	}
}

# Evaluated from a region scope. This deliberately iterates over every location
# in the map-data region; no owner or controller filter is applied.
eap_region_total_wealth = {
	value = 0
	every_location_in_region = {
		limit = { is_land = yes }
		add = eap_location_wealth
	}
}

# Evaluated from a region scope. Development from every land location is
# included without checking its owner or controller.
eap_region_total_development = {
	value = 0
	every_location_in_region = {
		limit = { is_land = yes }
		add = development
	}
}

# Evaluated when the invariant location-count cache is initialized at game start
# or rebuilt on save load, then copied to every land location for tooltips.
eap_region_total_land_locations = {
	value = 0
	every_location_in_region = {
		limit = { is_land = yes }
		add = 1
	}
}

# The denominator is cached by the annual April scan. The guarded default is used
# only before the game-start initialization has completed.
eap_region_stats_world_population = {
	# Population script values are measured in thousands, so this represents
	# 350 million people.
	value = 350000
	if = {
		limit = {
			has_global_variable = eap_region_stats_world_population
			global_var:eap_region_stats_world_population > 0
		}
		value = global_var:eap_region_stats_world_population
	}
	min = 0.01
}

# Returned as a fraction so the localization percentage formatter can multiply it
# by 100. Both numerator and denominator come from the same annual snapshot.
eap_region_population_world_share = {
	value = eap_region_population_map_value
	divide = eap_region_stats_world_population
	min = 0
	max = 1
}

# Annual regional snapshots used by maps and tooltips. The refresh copies each
# region's values to its land locations, so these are direct local cache reads.
# Population remains in the engine's native thousands-of-people unit.
eap_region_population_map_value = {
	value = var:eap_region_stats_cached_population
}

# Population script values use thousands; divide by 1,000 for tooltip millions.
eap_region_population_display_value = {
	value = eap_region_population_map_value
	divide = 1000
}

eap_region_wealth_map_value = {
	value = var:eap_region_stats_cached_wealth
}

eap_region_development_map_value = {
	value = var:eap_region_stats_cached_development
}

eap_region_land_location_count = {
	value = var:eap_region_stats_cached_land_location_count
}

# Cached ordinal positions generated by the same sorted passes that select the
# color thresholds. Rank 1 is the region with the highest value.
eap_region_population_world_rank = {
	value = var:eap_region_stats_cached_population_rank
}

eap_region_wealth_world_rank = {
	value = var:eap_region_stats_cached_wealth_rank

eap_region_development_world_rank = {

}

eap_region_stats_world_rank_total = {
	value = global_var:eap_region_stats_rank_total
}
